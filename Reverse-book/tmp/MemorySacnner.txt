#include <vector>
#include <windows.h>
#include <psapi.h>

// 一个简单的内存扫描函数
// 在指定进程的内存中查找所有等于 targetValue 的地址
std::vector<uintptr_t> SimpleMemoryScanner(HANDLE hProcess, int targetValue) {
    std::vector<uintptr_t> foundAddresses;

    // 1. 获取系统信息，确定扫描范围
    SYSTEM_INFO sysInfo;
    GetSystemInfo(&sysInfo);
    uintptr_t minAddress = (uintptr_t)sysInfo.lpMinimumApplicationAddress;
    uintptr_t maxAddress = (uintptr_t)sysInfo.lpMaximumApplicationAddress;

    MEMORY_BASIC_INFORMATION mbi;
    uintptr_t currentAddress = minAddress;

    // 2. 开始遍历内存
    while (currentAddress < maxAddress) {
        // 使用 VirtualQueryEx 查询当前地址的内存信息
        if (VirtualQueryEx(hProcess, (LPCVOID)currentAddress, &mbi, sizeof(mbi)) == 0) {
            break; // 查询失败，退出
        }

        // 3. 筛选可读的、已提交的、私有的内存页
        // 这可以跳过大量的系统保留区和不可访问的内存
        if ((mbi.State == MEM_COMMIT) &&
            (mbi.Protect & (PAGE_READWRITE | PAGE_READONLY | PAGE_EXECUTE_READ | PAGE_EXECUTE_READWRITE)) &&
            !(mbi.Protect & PAGE_GUARD) &&
            (mbi.Type == MEM_PRIVATE)) {

            // 4. 读取这块内存的数据
            std::vector<char> buffer(mbi.RegionSize);
            SIZE_T bytesRead;
            if (ReadProcessMemory(hProcess, mbi.BaseAddress, buffer.data(), mbi.RegionSize, &bytesRead)) {
                // 5. 在我们读取到的数据中查找目标值
                for (SIZE_T i = 0; i < bytesRead - sizeof(int); i++) {
                    int* potentialValue = (int*)(buffer.data() + i);
                    if (*potentialValue == targetValue) {
                        // 找到了！计算并保存真实地址
                        uintptr_t foundAddress = (uintptr_t)mbi.BaseAddress + i;
                        foundAddresses.push_back(foundAddress);
                    }
                }
            }
        }

        // 6. 移动到下一块内存区域
        currentAddress = (uintptr_t)mbi.BaseAddress + mbi.RegionSize;
    }

    return foundAddresses;
}
